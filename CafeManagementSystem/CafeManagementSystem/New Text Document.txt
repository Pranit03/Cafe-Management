-- Create the users table to handle authentication and roles
CREATE TABLE users (
    user_id INT PRIMARY KEY IDENTITY(1,1), -- Auto-incrementing primary key
    username VARCHAR(50) UNIQUE NOT NULL,    -- The login username, must be unique
    full_name VARCHAR(100) NOT NULL,         -- The full name of the user for display
    password_hash VARCHAR(255) NOT NULL,   -- IMPORTANT: This will store a HASH of the password, not the password itself
    role VARCHAR(20) NOT NULL,               -- Role of the user, e.g., 'Admin' or 'Staff'
    is_active BIT NOT NULL DEFAULT 1         -- To disable accounts without deleting them
);
GO

-- Create the orders table if it doesn't exist
IF NOT EXISTS (SELECT * FROM sysobjects WHERE name='orders' and xtype='U')
BEGIN
    CREATE TABLE orders (
        order_id INT PRIMARY KEY IDENTITY(1,1),
        user_id INT FOREIGN KEY REFERENCES users(user_id),
        order_date DATETIME NOT NULL DEFAULT GETDATE(),
        total_amount DECIMAL(10, 2) NOT NULL
    );
END
GO

-- Create the menu_items table if it doesn't exist
IF NOT EXISTS (SELECT * FROM sysobjects WHERE name='menu_items' and xtype='U')
BEGIN
    CREATE TABLE menu_items (
        item_id INT PRIMARY KEY IDENTITY(1,1),
        item_name VARCHAR(100) NOT NULL,
        category VARCHAR(50) NOT NULL,
        price DECIMAL(8, 2) NOT NULL
    );
END
GO
